const express = require("express"); const crypto = require("crypto"); const path = require("path"); const app = express(); const PORT = process.env.PORT || 3000; // Mobilde yapıştırırken boşluk kalma ihtimaline karşı .trim() eklendi const LINKVERTISE_TOKEN = (process.env.LINKVERTISE_TOKEN || "").trim(); const KEY_DURATION_MS = 24 * 60 * 60 * 1000; // 24 saat const activeKeys = new Map(); function generateKey() { const part = () => crypto.randomBytes(2).toString("hex").toUpperCase(); return `SPECTRE-${part()}-${part()}-${part()}`; } async function verifyLinkvertiseHash(hash, res) { console.log("[VERIFY] Gelen Hash:", hash || "YOK (Linkvertise Anti-Bypass kapalı veya kaydedilmemiş)"); console.log("[VERIFY] Token Uzunluğu:", LINKVERTISE_TOKEN.length); if (!LINKVERTISE_TOKEN || LINKVERTISE_TOKEN.length < 32) { console.log("[VERIFY] HATA: Render Environment içinde LINKVERTISE_TOKEN eksik!"); return res.redirect("/?error=bypass"); } if (!hash || typeof hash !== "string" || hash.length < 16) { return res.redirect("/?error=bypass"); } try { const verifyUrl = `https://publisher.linkvertise.com/api/v1/anti_bypassing?token=${encodeURIComponent( LINKVERTISE_TOKEN )}&hash=${encodeURIComponent(hash.trim())}`; const response = await fetch(verifyUrl, { method: "POST" }); const result = (await response.text()).trim(); console.log("[VERIFY] Linkvertise API Yanıtı:", result); if (result !== "TRUE") { return res.redirect("/?error=bypass"); } const newKey = generateKey(); activeKeys.set(newKey, { expiresAt: Date.now() + KEY_DURATION_MS }); return res.redirect(`/?key=${encodeURIComponent(newKey)}`); } catch (err) { console.error("[VERIFY] API Bağlantı Hatası:", err); return res.redirect("/?error=bypass"); } } app.use(express.static(path.join(__dirname, "public"))); // Hem ana sayfaya (/?hash=...) hem /claim (/claim?hash=...) yönlendirmelerini yakalar app.get("/", async (req, res) => { if (req.query.hash) { return verifyLinkvertiseHash(req.query.hash, res); } return res.sendFile(path.join(__dirname, "public", "index.html")); }); app.get("/claim", async (req, res) => { return verifyLinkvertiseHash(req.query.hash, res); }); // Roblox Lua scripti için key doğrulama rotası app.get("/api/verify", (req, res) => { const { key } = req.query; if (!key || !activeKeys.has(key)) { return res.json({ valid: false, message: "Invalid key" }); } const record = activeKeys.get(key); if (Date.now() > record.expiresAt) { activeKeys.delete(key); return res.json({ valid: false, message: "Key expired" }); } return res.json({ valid: true, expiresAt: record.expiresAt }); }); app.listen(PORT, () => { console.log(`Spectre Hub Key System aktif — Port: ${PORT}`); });
